\begin{lemma}[Rational Gap Example for Suprema]
\label{lem:rational-gap-suprema}
Let
\[
  A=\{q\in\mathbb{Q}:q^2<2\}.
\]
Then $A$ is nonempty and bounded above relative to $\mathbb{Q}$, but $A$ has
no supremum in $\mathbb{Q}$:
\[
  \nexists s\in\mathbb{Q}\left[
  (\forall q\in A)(q\leq s)
  \land
  \forall u\in\mathbb{Q}
  \bigl((\forall q\in A)(q\leq u)\Longrightarrow s\leq u\bigr)
  \right].
\]
When $A$ is regarded as a subset of $\mathbb{R}$, its supremum exists and is
\[
  \sup_{\mathbb{R}}A=\sqrt{2}.
\]

\end{lemma}
