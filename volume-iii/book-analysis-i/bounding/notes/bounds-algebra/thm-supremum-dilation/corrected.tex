\begin{theorem}[Supremum of a Dilation]
\label{thm:supremum-dilation}
Let $A\subseteq\mathbb{R}$ be nonempty and bounded, and let
$\lambda\in\mathbb{R}$.  Then $\lambda A:=\{\lambda a:a\in A\}$ is nonempty
and bounded above, and
\[
  \sup(\lambda A)=
  \begin{cases}
    \lambda\sup A, & \lambda>0,\\
    0, & \lambda=0,\\
    \lambda\inf A, & \lambda<0.
  \end{cases}
\]

\end{theorem}
