\begin{lemma}[Irrational Plus Rational Is Irrational]
\label{lem:irrational-plus-rational-is-irrational}
For every $x\in\mathbb{R}\setminus\mathbb{Q}$ and every $q\in\mathbb{Q}$,
\[
  x+q\in\mathbb{R}\setminus\mathbb{Q}.
\]

\end{lemma}
