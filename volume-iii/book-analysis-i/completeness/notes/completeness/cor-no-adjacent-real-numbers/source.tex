\begin{corollary}[No Adjacent Real Numbers]
\label{cor:no-adjacent-real-numbers}
For every $a,b\in\mathbb{R}$, if $a<b$, then there exists
$c\in\mathbb{R}$ such that
\[
  a<c<b.
\]
\hyperref[prf:no-adjacent-real-numbers]{Go to proof.}
\end{corollary}
