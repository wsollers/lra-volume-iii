\begin{theorem}[Archimedean Property]
\label{thm:archimedean-property}
For every $x,y\in\mathbb{R}$, if $x>0$ and $y>0$, then there exists
$n\in\mathbb{N}$ such that
\[
  nx>y.
\]
Equivalently,
\[
  (\forall x,y\in\mathbb{R})
  \bigl[(x>0\land y>0)\Longrightarrow(\exists n\in\mathbb{N})(nx>y)\bigr].
\]

\end{theorem}
