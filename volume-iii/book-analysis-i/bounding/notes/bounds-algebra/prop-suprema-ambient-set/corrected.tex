\begin{proposition}[Suprema Depend on the Ambient Set]
\label{prop:suprema-ambient-set}
There exist a linearly ordered set $(P,\leq)$, subsets
$A\subseteq S\subseteq S'\subseteq P$, and relative suprema $\sup_S A$ and
$\sup_{S'}A$ such that
\[
  \sup_S A\ne\sup_{S'}A.
\]
For example, in $\mathbb{R}$, let
\[
  A=(0,1),\qquad
  S=(0,1)\cup\{2\},
  \qquad
  S'=(0,1)\cup\left\{\frac{3}{2},2\right\}.
\]
Then
\[
  \sup_S A=2,
  \qquad
  \sup_{S'}A=\frac{3}{2}.
\]

\end{proposition}
