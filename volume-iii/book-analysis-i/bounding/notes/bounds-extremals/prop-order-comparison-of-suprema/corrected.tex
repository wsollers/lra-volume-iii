\begin{proposition}[Order Comparison of Suprema]
\label{prop:order-comparison-of-suprema}
Let $A,B\subseteq\mathbb{R}$ be nonempty and bounded above. Suppose that
\[
  (\forall a\in A)(\exists b\in B)(a\leq b).
\]
Then
\[
  \sup A\leq\sup B.
\]

\end{proposition}
