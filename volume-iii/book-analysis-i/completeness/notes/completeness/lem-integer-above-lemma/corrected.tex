\begin{lemma}[Integer Above Lemma]
\label{lem:integer-above-lemma}
For every $x\in\mathbb{R}$, there exists $n\in\mathbb{N}$ such that
\[
  x<n.
\]
Equivalently,
\[
  (\forall x\in\mathbb{R})(\exists n\in\mathbb{N})(x<n).
\]
