\begin{proposition}[Upper Bounds Depend on the Ambient Order]
\label{prop:upper-bounds-ambient-order}
Let $(P,\leq)$ be a partially ordered set, and let
$A\subseteq S\subseteq S'\subseteq P$.  The upper bounds of $A$ relative to
$S$ are exactly those upper bounds of $A$ relative to $S'$ that belong to $S$:
\[
  \{u\in S:(\forall a\in A)(a\leq u)\}
  =
  S\cap\{u\in S':(\forall a\in A)(a\leq u)\}.
\]

\end{proposition}
