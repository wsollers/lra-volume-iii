\begin{lemma}[Irrational Minus Rational Is Irrational]
\label{lem:irrational-minus-rational-is-irrational}
For every $x\in\mathbb{R}\setminus\mathbb{Q}$ and every $q\in\mathbb{Q}$,
\[
  x-q\in\mathbb{R}\setminus\mathbb{Q}.
\]

\end{lemma}
