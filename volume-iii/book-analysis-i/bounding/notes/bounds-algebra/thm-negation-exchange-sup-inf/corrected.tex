\begin{theorem}[Negation Exchanges Supremum and Infimum]
\label{thm:negation-exchange-sup-inf}
Let $A\subseteq\mathbb{R}$ be nonempty and bounded below.  Then
$-A:=\{-a:a\in A\}$ is nonempty and bounded above, and
\[
  \sup(-A)=-\inf A.
\]

\end{theorem}
