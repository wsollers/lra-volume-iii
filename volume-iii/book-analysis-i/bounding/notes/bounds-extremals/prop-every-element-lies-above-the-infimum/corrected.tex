\begin{proposition}[Every Element Lies Above the Infimum]
\label{prop:every-element-lies-above-the-infimum}
Let $A\subseteq\mathbb{R}$ be nonempty and bounded below, and let
$i=\inf A$. Then
\[
  (\forall x\in A)(i\leq x).
\]

\end{proposition}
