\begin{lemma}[Rational Minus Irrational Is Irrational]
\label{lem:rational-minus-irrational-is-irrational}
For every $q\in\mathbb{Q}$ and every $x\in\mathbb{R}\setminus\mathbb{Q}$,
\[
  q-x\in\mathbb{R}\setminus\mathbb{Q}.
\]

\end{lemma}
