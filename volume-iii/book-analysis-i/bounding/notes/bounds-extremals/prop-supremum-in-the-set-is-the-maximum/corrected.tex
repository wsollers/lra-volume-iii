\begin{proposition}[Supremum in the Set Is the Maximum]
\label{prop:supremum-in-the-set-is-the-maximum}
Let $A\subseteq\mathbb{R}$ be nonempty and bounded above, and let
$s=\sup A$. If $s\in A$, then
\[
  s=\max A.
\]

\end{proposition}
