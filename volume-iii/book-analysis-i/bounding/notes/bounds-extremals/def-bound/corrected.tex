\begin{definition}[Bound]
\label{def:bound}
Let $(S,\leq)$ be a partially ordered set, let $A\subseteq S$, and let
$b\in S$. The element $b$ is a \emph{bound} of $A$ if it is either an
upper bound or a lower bound of $A$; equivalently,
\[
  \bigl[(\forall a\in A)(a\leq b)\bigr]
  \lor
  \bigl[(\forall a\in A)(b\leq a)\bigr].
\]
\end{definition}
