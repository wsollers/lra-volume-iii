\begin{proposition}[Every Element Lies Below the Supremum]
\label{prop:every-element-lies-below-the-supremum}
Let $A\subseteq\mathbb{R}$ be nonempty and bounded above, and let
$s=\sup A$. Then
\[
  (\forall x\in A)(x\leq s).
\]

\end{proposition}
