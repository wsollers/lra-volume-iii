\begin{proposition}[Bounds for Suprema and Infima Under Set Addition]
\label{prop:bounds-sum-set}
Let $A,B\subseteq\mathbb{R}$, and let
$u_A,u_B,\ell_A,\ell_B\in\mathbb{R}$.  If $u_A$ is an upper bound of $A$ and
$u_B$ is an upper bound of $B$, then $u_A+u_B$ is an upper bound of $A+B$:
\[
  \left[(\forall a\in A)(a\leq u_A)\land
  (\forall b\in B)(b\leq u_B)\right]
  \Longrightarrow
  (\forall x\in A+B)(x\leq u_A+u_B).
\]
If $\ell_A$ is a lower bound of $A$ and $\ell_B$ is a lower bound of $B$, then
$\ell_A+\ell_B$ is a lower bound of $A+B$:
\[
  \left[(\forall a\in A)(\ell_A\leq a)\land
  (\forall b\in B)(\ell_B\leq b)\right]
  \Longrightarrow
  (\forall x\in A+B)(\ell_A+\ell_B\leq x).
\]

\end{proposition}
