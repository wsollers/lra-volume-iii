\begin{theorem}[Supremum of a Difference Set]
\label{thm:supremum-difference-set}
Let $A,B\subseteq\mathbb{R}$ be nonempty, with $A$ bounded above and $B$
bounded below.  Then $A-B:=\{a-b:a\in A,\ b\in B\}$ is nonempty and bounded
above, and
\[
  \sup(A-B)=\sup A-\inf B.
\]

\end{theorem}
