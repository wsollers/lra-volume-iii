\begin{theorem}[Ambient Existence of Supremum]
\label{thm:ambient-existence-supremum}
There exist a linearly ordered set $(P,\leq)$ and subsets
$A\subseteq S\subseteq S'\subseteq P$ such that $A$ has a supremum relative to
$S'$ but has no supremum relative to $S$.  For example, in $\mathbb{R}$, let
\[
  A=S=(0,1),
  \qquad
  S'=(0,1].
\]
Then
\[
  \sup_{S'}A=1,
\]
whereas $A$ has no upper bound in $S$, and hence no supremum relative to $S$.

\end{theorem}
